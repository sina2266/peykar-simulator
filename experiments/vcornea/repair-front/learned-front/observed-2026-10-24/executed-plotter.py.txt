"""Read retained observations/predictions only; never calibrate or execute models."""
import json
from pathlib import Path
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt
base=Path('experiments/vcornea/repair-front/learned-front')
observed=base/'observed-2026-10-24'
r=json.loads((observed/'report.json').read_bytes())
fig,axes=plt.subplots(1,3,figsize=(16,5.3),constrained_layout=True)
color='#176c8e'
for case,ax in zip(['heldout2017','challenge2021','fischak2017'],axes):
 d=json.loads((observed/case/'candidate.json').read_bytes())
 y=next(o['values'] for o in d['outputs'] if o['name']=='open_area_fraction')
 ax.plot([t/3600 for t in d['time']['values']],[100*a for a in y],color='black',lw=2,label='New learned prediction')
 ax.set(xlabel='Hours after debridement',ylabel='Open projected area (% of initial)',ylim=(-2,110),xlim=(0,max(d['time']['values'])/3600))
 ax.grid(alpha=.15)
data=json.loads((base/'data/heldout.json').read_bytes())
for animal in ['1','2','3']:
 rows=[v for v in data if v['replicate']==animal]
 axes[0].plot([v['hour'] for v in rows],[100*v['wound_area_fraction'] for v in rows],color=color,alpha=.35,lw=.9)
rows=r['heldout2017']['rows'];hours=[v['hour'] for v in rows]
axes[0].plot([0]+hours,[100]+[100*v['measured_mean_fraction'] for v in rows],'o',color=color,label='Measured mean (3 excluded rabbits)')
axes[0].plot(hours,[100*v['unchanged1986_baseline_fraction'] for v in rows],'--',color='#888888',label='Unchanged 1986 comparator')
axes[0].set_title('Retrospective 0.5%SDP animal/dose test\nRMSE 5.96 pp / max 12.26 pp: FAIL',loc='left',fontsize=11)
for group,c,marker in zip(r['challenge2021']['groups'],['#ad4c32','#286c54','#72518d'],['o','s','^']):
 rows=group['rows']
 axes[1].plot([v['hour'] for v in rows],[100*v['mean_fraction'] for v in rows],marker=marker,lw=1,color=c,label=group['group'])
axes[1].text(13,8,'50h Ocu/Amo: categorical healed;\nno fabricated numeric zeros',fontsize=8)
axes[1].set_title('Independent 2021 removal methods\nAll three groups fail; RMSE 28.28–29.46 pp',loc='left',fontsize=11)
rows=r['fischak2017']['rows'];hours=[v['hour'] for v in rows];bounds=[100*v['measured_ratio_lower_constraint'] for v in rows]
axes[2].plot(hours,bounds,'^',color='#ad4c32',label='Conservative lower constraints only')
for t,b in zip(hours,bounds):
 axes[2].annotate('',xy=(t,b+11),xytext=(t,b),arrowprops={'arrowstyle':'->','color':'#ad4c32','lw':1})
axes[2].text(18,82,'Means and animal IDs not recovered.\nArrows indicate one-sided constraints;\nnot measured means or confidence intervals.',fontsize=8)
axes[2].set_title('New independent Fischak2017 PBS study\nOptimistic RMSE lower bound 5.57 pp: FAIL',loc='left',fontsize=11)
axes[2].set_xlim(0,74)
for ax in axes:ax.legend(loc='upper right',fontsize=7,frameon=False)
fig.suptitle('Frozen 5 pp RMSE / 10 pp maximum targets unchanged — fitting is not biological validation',fontsize=12)
fig.savefig(observed/'learned-predictions.png',dpi=160)
fig.savefig(observed/'learned-predictions.pdf')
