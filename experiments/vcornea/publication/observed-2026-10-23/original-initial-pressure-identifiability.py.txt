import hashlib,importlib.util,json,math
from fractions import Fraction
from pathlib import Path

def recover(rows):
 if not rows:raise ValueError('Actual original cell observations required')
 points=[]
 for r in rows:
  v,p=r['volume'],r['absolute_pressure']
  if not math.isfinite(v) or not math.isfinite(p) or v<=0 or p<0:raise ValueError('Invalid original volume/absolute pressure')
  points.append((Fraction(str(v)),Fraction(str(p))))
 distinct=next(((v,p) for v,p in points[1:] if v!=points[0][0]),None)
 if distinct is None:return {'status':'not_identifiable' if len({p for v,p in points})==1 else 'no_shared_solution','solutions':[]}
 v1,p1=points[0];v2,p2=distinct;solutions=set()
 for s1 in [-1,1]:
  for s2 in [-1,1]:
   a=(s1*p1-s2*p2)/(v1-v2)
   if a<=0:continue
   target=v1-s1*p1/a
   if target<0:continue
   if all(abs(a*(v-target))==p for v,p in points):solutions.add((a/2,target))
 return {'status':'unique_under_declared_shared_parameter_model' if len(solutions)==1 else 'ambiguous' if solutions else 'no_shared_solution','solutions':[{'lambda_volume':float(lam),'lambda_exact_fraction':str(lam),'initial_target_volume':float(target),'target_exact_fraction':str(target)} for lam,target in sorted(solutions)],'cells_checked':len(rows),'distinct_observed_volumes':len({v for v,p in points}),'exact_arithmetic':True,'outliers_discarded':0}
